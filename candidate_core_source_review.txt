 169: def build_high_conviction_buy_gate(

 174: ) -> tuple[HighConvictionDecision, ...]:
 175:     """Classify existing daily decisions without changing the AL/SAT engine.
 176: 
 177:     This is a strict confidence overlay. It never upgrades BEKLE/İŞLEM YOK to AL,
 178:     never changes existing thresholds, and creates no combined confidence score.
 179:     """
 180:     cards_by_symbol = {
 181:         str(getattr(card, "symbol", "")).strip().upper(): card for card in cards
 182:     }
 183:     calibration = tuple(calibration_slices)
 184:     result: list[HighConvictionDecision] = []
 185: 
 186:     for decision in decisions:
 187:         symbol = str(getattr(decision, "symbol", "")).strip().upper()
 188:         card = cards_by_symbol.get(symbol)
 189:         base = str(getattr(decision, "decision", "")).upper()
 190:         priority = int(getattr(decision, "priority", 0))
 191: 
 192:         score = int(getattr(decision, "technical_score", 0))
 193:         view = str(getattr(decision, "technical_view", ""))
 194:         volume = float(getattr(decision, "volume_ratio", 0.0))
 195:         rr = float(getattr(decision, "risk_reward", 0.0))
 196:         data_status = str(getattr(decision, "data_status", "UYGUN") or "UYGUN").upper()
 197: 
 198:         technical_ok = _positive_technical(view) and score >= 80
 199:         volume_ok = volume >= 1.30
 200:         rr_ok = rr >= 2.0
 201:         timing_ok, timing_reason = _timing_pass(card, decision)
 202:         entry_ok, entry_reason = _entry_pass(card) if card is not None else (False, "Aday kartı yok")
 203:         data_ok = data_status == "UYGUN"
 204:         history = evaluate_historical_support(decision, calibration)
 205:         history_ok = history.status == "DESTEKLEYİCİ"
 206: 
 207:         reasons: list[str] = []
 208:         blockers: list[str] = []
 209: 
 210:         if technical_ok:
 211:             reasons.append(f"Teknik kapı geçti ({score})")
 212:         else:
 213:             blockers.append(f"Teknik kapı: pozitif görünüm + skor >=80 gerekli ({score})")
 214:         if volume_ok:
 215:             reasons.append(f"Hacim kapısı geçti ({volume:.2f})")
 216:         else:
 217:             blockers.append(f"Hacim kapısı: >=1.30 gerekli ({volume:.2f})")
 218:         if rr_ok:
 219:             reasons.append(f"Risk/getiri kapısı geçti ({rr:.2f})")
 220:         else:
 221:             blockers.append(f"Risk/getiri kapısı: >=2.00 gerekli ({rr:.2f})")
 222:         if timing_ok:
 223:             reasons.append(timing_reason)
 224:         else:
 225:             blockers.append(timing_reason)
 226:         if entry_ok:
 227:             reasons.append(entry_reason)
 228:         else:
 229:             blockers.append(entry_reason)
 230:         if data_ok:
 231:             reasons.append("Veri durumu UYGUN")
 232:         else:
 233:             blockers.append("Kısmi/gecikmeli veri yüksek güveni engelliyor")
 234:         if history_ok:
 235:             reasons.append("Outcome geçmişi destekleyici")
 236:             reasons.extend(history.reasons)
 237:         else:
 238:             blockers.append(
 239:                 "Outcome geçmişi yüksek güven için henüz destekleyici değil "

 241:             )
 242: 
 243:         if base != "AL":
 244:             conviction = NO_TRADE if base == NO_TRADE else WAIT
 245:         elif all((technical_ok, volume_ok, rr_ok, timing_ok, entry_ok, data_ok, history_ok)):

 248:             conviction = NORMAL_CONFIDENCE

 251:             HighConvictionDecision(

 253:                 symbol=symbol,

 260:                 entry_gate="GEÇTİ" if entry_ok else "KALDI",
 261:                 data_gate="GEÇTİ" if data_ok else "KALDI",
 262:                 history_gate=history.status,
 263:                 reasons=tuple(dict.fromkeys(reasons)),
 264:                 blockers=tuple(dict.fromkeys(blockers)),
 265:             )
 266:         )
