class DataSourceRecoveryError(Exception):
    """Birincil, ikincil ve önbellek kaynaklarının tamamı başarısız olduğunda kullanılır."""


@dataclass(frozen=True)
class RecoveredMarketData:
    veri: pd.DataFrame
    kaynak: str
    kaynak_sirasi: tuple[str, ...]
    kaynak_hatalari: tuple[str, ...]


SecondaryLoader = Callable[[str], pd.DataFrame]
FreshnessEvaluator = Callable[[pd.DataFrame], object]


def _cache_paths(cache_root: Path, symbol: str) -> tuple[Path, Path]:
    safe = symbol.replace("^", "_").replace(".", "_").replace("/", "_")
    return cache_root / f"{safe}.csv", cache_root / f"{safe}.json"


def cache_write(cache_root: Path, symbol: str, frame: pd.DataFrame) -> None:
    cache_root.mkdir(parents=True, exist_ok=True)
    csv_path, meta_path = _cache_paths(cache_root, symbol)
    normalized = fiyat_verisini_duzelt(frame, yalnizca_tamamlanmis_gunler=True)
    csv_text = normalized.to_csv(index=True, lineterminator="\n")
    csv_path.write_text(csv_text, encoding="utf-8")
    meta = {
        "symbol": symbol,
        "rows": len(normalized),
        "first_date": str(pd.Timestamp(normalized.index[0]).date()),
        "last_date": str(pd.Timestamp(normalized.index[-1]).date()),
        "sha256": hashlib.sha256(csv_text.encode("utf-8")).hexdigest(),
        "written_at": datetime.now(timezone.utc).isoformat(),
    }
    meta_path.write_text(json.dumps(meta, ensure_ascii=False, indent=2), encoding="utf-8")


def cache_read(cache_root: Path, symbol: str) -> pd.DataFrame:
    csv_path, meta_path = _cache_paths(cache_root, symbol)
    if not csv_path.exists() or not meta_path.exists():
        raise DataSourceRecoveryError("Yerel önbellek kaydı bulunamadı.")
    csv_text = csv_path.read_text(encoding="utf-8")
    meta = json.loads(meta_path.read_text(encoding="utf-8"))
    if hashlib.sha256(csv_text.encode("utf-8")).hexdigest() != meta.get("sha256"):
        raise DataSourceRecoveryError("Yerel önbellek bütünlük doğrulamasını geçemedi.")
    frame = pd.read_csv(io.StringIO(csv_text), index_col=0, parse_dates=True)
    frame = fiyat_verisini_duzelt(frame, yalnizca_tamamlanmis_gunler=True)
    if len(frame) != int(meta.get("rows", -1)):
        raise DataSourceRecoveryError("Yerel önbellek satır sayısı doğrulanamadı.")
    freshness = gunluk_veri_tazeligini_degerlendir(frame)
    if freshness.kilitli:
        raise DataSourceRecoveryError("Yerel önbellek güncel değil: " + freshness.aciklama)
    return frame


def alpha_vantage_daily_loader(symbol: str) -> pd.DataFrame:
    api_key = os.getenv("HISSE_ROBOTU_ALPHA_VANTAGE_API_KEY", "").strip()
    if not api_key:
        raise DataSourceRecoveryError("Alpha Vantage API anahtarı tanımlı değil.")
    raw_map = os.getenv("HISSE_ROBOTU_ALPHA_VANTAGE_SYMBOL_MAP", "{}")
    try:
        symbol_map = json.loads(raw_map)
    except json.JSONDecodeError as exc:
        raise DataSourceRecoveryError("Alpha Vantage sembol eşleme JSON'u geçersiz.") from exc
    provider_symbol = str(symbol_map.get(symbol, symbol))
    query = urlencode({
        "function": "TIME_SERIES_DAILY",
        "symbol": provider_symbol,
        "outputsize": "full",
        "datatype": "csv",
        "apikey": api_key,
    })
    try:
        with urlopen("https://www.alphavantage.co/query?" + query, timeout=30) as response:
            payload = response.read()
    except Exception as exc:
        raise DataSourceRecoveryError(f"Alpha Vantage bağlantı hatası: {exc}") from exc
    frame = pd.read_csv(io.BytesIO(payload))
    required = {"timestamp", "open", "high", "low", "close", "volume"}
    if not required.issubset(frame.columns):
        message = payload.decode("utf-8", errors="replace")[:300]
        raise DataSourceRecoveryError("Alpha Vantage beklenen OHLCV yanıtını üretmedi: " + message)
    frame = frame.rename(columns={
        "timestamp": "Date", "open": "Open", "high": "High",
        "low": "Low", "close": "Close", "volume": "Volume",
    })
    frame["Date"] = pd.to_datetime(frame["Date"])
    return fiyat_verisini_duzelt(
        frame.set_index("Date").sort_index(),
        yalnizca_tamamlanmis_gunler=True,
    )


def recover_daily_market_data(
    symbol: str,
    *,
    primary_loader: Callable[[], pd.DataFrame],
    secondary_loader: SecondaryLoader | None = None,
    cache_root: Path | None = None,
    freshness_evaluator: FreshnessEvaluator | None = None,
) -> RecoveredMarketData:
    cache_root = cache_root or Path("backend/cache/market_data")
    evaluator = freshness_evaluator or gunluk_veri_tazeligini_degerlendir
    order: list[str] = []
    failures: list[str] = []

    order.append("birincil")
    try:
        frame = fiyat_verisini_duzelt(primary_loader(), yalnizca_tamamlanmis_gunler=True)
        freshness = evaluator(frame)
        if freshness.kilitli:
            raise DataSourceRecoveryError(freshness.aciklama)
        cache_write(cache_root, symbol, frame)
        return RecoveredMarketData(frame, "birincil", tuple(order), tuple(failures))
    except Exception as exc:
        failures.append(f"birincil: {exc}")

    order.append("ikincil")
    try:
        frame = (secondary_loader or alpha_vantage_daily_loader)(symbol)
        freshness = evaluator(frame)
        if freshness.kilitli:
            raise DataSourceRecoveryError(freshness.aciklama)
        cache_write(cache_root, symbol, frame)
        return RecoveredMarketData(frame, "ikincil", tuple(order), tuple(failures))
    except Exception as exc:
        failures.append(f"ikincil: {exc}")

    order.append("önbellek")
    try:
        frame = cache_read(cache_root, symbol)
        return RecoveredMarketData(frame, "önbellek", tuple(order), tuple(failures))
    except Exception as exc:
        failures.append(f"önbellek: {exc}")

    raise DataSourceRecoveryError("Tüm veri kaynakları başarısız. " + " | ".join(failures))

@dataclass(frozen=True)
class ScanFailure:
    """Analiz edilemeyen bir hisseyi temsil eder."""

    sembol: str
    hata: str


@dataclass(frozen=True)
class StockScanResult:
    """Çoklu hisse taramasının bütün sonucunu temsil eder."""

    basarili_sonuclar: tuple[StockAnalysisResult, ...]
    hatalar: tuple[ScanFailure, ...]

    @property
    def toplam_sembol(self) -> int:
        return len(self.basarili_sonuclar) + len(self.hatalar)

    @property
    def basarili_sayisi(self) -> int:
        return len(self.basarili_sonuclar)

    @property
    def hatali_sayisi(self) -> int:
        return len(self.hatalar)

    @property
    def pozitif_sayisi(self) -> int:
        return sum(
            sonuc.sinyal.karar == "Pozitif Teknik Görünüm"
            for sonuc in self.basarili_sonuclar
        )

    @property
    def notr_sayisi(self) -> int:
        return sum(
            sonuc.sinyal.karar == "Nötr Teknik Görünüm"
            for sonuc in self.basarili_sonuclar
        )

    @property
    def negatif_sayisi(self) -> int:
        return sum(
            sonuc.sinyal.karar == "Negatif Teknik Görünüm"
            for sonuc in self.basarili_sonuclar
        )


class StockScanner:
    """
    Birden fazla hisseyi analiz eder ve sonuçları
    sinyal skoruna göre sıralar.
    """

    def __init__(
        self,
        analiz_motoru: StockAnalyzer | None = None,
    ) -> None:
        self._analiz_motoru = (
            analiz_motoru
            if analiz_motoru is not None
            else StockAnalyzer()
        )

    def tara(
        self,
        semboller: Iterable[str],
        ilerleme_bildirimi: (
            Callable[[int, int, str], None] | None
        ) = None,
    ) -> StockScanResult:
        """
        Verilen sembolleri sırayla analiz eder.

        Nokta içermeyen sembollere BIST için otomatik olarak
        '.IS' uzantısı eklenir.
        """
        temiz_semboller = _sembolleri_hazirla(
            semboller
        )

        if not temiz_semboller:
            raise ValueError(
                "Taranacak en az bir hisse sembolü gereklidir."
            )

        basarili_sonuclar: list[StockAnalysisResult] = []
        hatalar: list[ScanFailure] = []

        toplam = len(temiz_semboller)

        for sira, sembol in enumerate(
            temiz_semboller,
            start=1,
        ):
            if ilerleme_bildirimi is not None:
                ilerleme_bildirimi(
                    sira,
                    toplam,
                    sembol,
                )

            try:
                sonuc = self._analiz_motoru.analiz_et(
                    sembol
                )

            except (
                ValueError,
                StockAnalyzerError,
            ) as hata:
                hatalar.append(
                    ScanFailure(
                        sembol=sembol,
                        hata=str(hata),
                    )
                )
                continue

            basarili_sonuclar.append(sonuc)

        sirali_sonuclar = sorted(
            basarili_sonuclar,
            key=lambda sonuc: (
                -sonuc.sinyal.skor,
                -sonuc.risk_plani.risk_odul_orani,
                sonuc.sembol,
            ),
        )

        return StockScanResult(
            basarili_sonuclar=tuple(
                sirali_sonuclar
            ),
            hatalar=tuple(hatalar),
        )


class BatchStockScanner:
    """
    Birden fazla hisseyi tek toplu veri indirmesi üzerinden analiz eder.

    StockScanner ile aynı sonuç modelini üretir; farkı, geniş BIST evrenlerinde
    her sembol için ayrı ağ çağrısı yapmak yerine veriyi parçalar halinde
    indirmesidir.
    """

    def __init__(
        self,
        analiz_motoru: StockAnalyzer | None = None,
        veri_getirici: Callable[..., BatchMarketDataResult] = (
            toplu_fiyat_verisi_getir
        ),
    ) -> None:
        self._analiz_motoru = analiz_motoru or StockAnalyzer()
        self._veri_getirici = veri_getirici

    def tara(
        self,
        semboller: Iterable[str],
        ilerleme_bildirimi: (
            Callable[[int, int, str], None] | None
        ) = None,
    ) -> StockScanResult:
        temiz_semboller = _sembolleri_hazirla(semboller)
        if not temiz_semboller:
            raise ValueError(
                "Taranacak en az bir hisse sembolü gereklidir."
            )

        veri_sonucu = self._veri_getirici(
            temiz_semboller,
            islem_gunu=160,
            indirme_donemi="1y",
        )
        veri_hatalari = {
            hata.sembol: hata.hata for hata in veri_sonucu.hatalar
        }

        basarili_sonuclar: list[StockAnalysisResult] = []
        hatalar: list[ScanFailure] = []
        toplam = len(temiz_semboller)

        for sira, sembol in enumerate(temiz_semboller, start=1):
            if ilerleme_bildirimi is not None:
                ilerleme_bildirimi(sira, toplam, sembol)

            fiyat_verisi = veri_sonucu.veriler.get(sembol)
            if fiyat_verisi is None:
                hatalar.append(
                    ScanFailure(
                        sembol=sembol,
                        hata=veri_hatalari.get(
                            sembol, "Toplu veride sembol bulunamadı."
                        ),
                    )
                )
                continue

            try:
                sonuc = self._analiz_motoru.veriden_analiz_et(
                    sembol=sembol,
                    fiyat_verisi=fiyat_verisi,
                )
            except (ValueError, StockAnalyzerError) as hata:
                hatalar.append(
                    ScanFailure(sembol=sembol, hata=str(hata))
                )
                continue

            basarili_sonuclar.append(sonuc)

        sirali_sonuclar = sorted(
            basarili_sonuclar,
            key=lambda sonuc: (
                -sonuc.sinyal.skor,
                -sonuc.risk_plani.risk_odul_orani,
                -sonuc.hacim_orani,
                sonuc.sembol,
            ),
        )

        return StockScanResult(
            basarili_sonuclar=tuple(sirali_sonuclar),
            hatalar=tuple(hatalar),
        )


def _sembolleri_hazirla(
    semboller: Iterable[str],
) -> list[str]:
    """
    Sembolleri temizler, BIST uzantısını ekler ve tekrarları siler.
    """
    sonuc: list[str] = []
    gorulenler: set[str] = set()

    for sembol in semboller:
        temiz_sembol = str(
            sembol
        ).lstrip("\ufeff").strip().upper()

        if not temiz_sembol:
            continue

        if "." not in temiz_sembol:
            temiz_sembol = (
                f"{temiz_sembol}.IS"
            )

        if temiz_sembol in gorulenler:
            continue

        gorulenler.add(temiz_sembol)
        sonuc.append(temiz_sembol)

    return sonuc

from __future__ import annotations

import os

from pathlib import Path
from typing import Any

from backend.app.analysis.market_scanner import (
    BistMarketScannerService,
)

    DashboardManualVbtsError,
    load_dashboard_vbts_dataset,
)

class DashboardMarketScannerSourceError(RuntimeError):
    """Dashboard tarama servisi seçilen VBTS kaynağıyla kurulamadığında oluşur."""


NowProvider = Callable[[], datetime]


def build_dashboard_market_scanner(
    *,
    now_provider: NowProvider | None = None,
    maximum_age: timedelta = timedelta(hours=24),
    evren_repository: Any = None,
    tarayici: Any = None,
    izleme_listesi_yolu: str | Path | None = None,
) -> BistMarketScannerService:
    """Seçili VBTS veri kaynağıyla korumalı tarama servisi oluşturur."""

    source_name = os.getenv(
        "HISSE_ROBOTU_VBTS_SOURCE",
        "",
    ).strip().lower()

    if not source_name:
        raise DashboardMarketScannerSourceError(
            "VBTS veri kaynağı ortam değişkeninde tanımlanmalıdır."
        )

    now = (
        now_provider()
        if now_provider is not None
        else datetime.now(timezone.utc)
    )

    if now.tzinfo is None:
        raise DashboardMarketScannerSourceError(
            "VBTS kontrol zamanı saat dilimi içermelidir."
        )

    if maximum_age <= timedelta(0):
        raise DashboardMarketScannerSourceError(
            "VBTS azami veri yaşı sıfırdan büyük olmalıdır."
        )

    if source_name != "manual_csv":
        raise DashboardMarketScannerSourceError(
            f"VBTS veri kaynağı desteklenmiyor: {source_name}"
        )

    try:
        dataset = load_dashboard_vbts_dataset(
            now_provider=lambda: now,
        )

        vbts_filter = VbtsScanFilter(
            dataset=dataset,
            maximum_age=maximum_age,
        )

        return BistMarketScannerService(
            evren_repository=evren_repository,
            tarayici=tarayici,
            izleme_listesi_yolu=izleme_listesi_yolu,
            vbts_scan_filter=vbts_filter,
            vbts_control_date=now.date(),
            vbts_control_time=now,
        )
    except DashboardManualVbtsError as exc:
        raise DashboardMarketScannerSourceError(str(exc)) from exc
    except Exception as exc:
        raise DashboardMarketScannerSourceError(
            f"VBTS korumalı tarama servisi hazırlanamadı: {exc}"
        ) from exc

from __future__ import annotations

import numpy as np

from backend.app.data.batch_market_data import toplu_fiyat_verisi_getir


def coklu_veri(semboller: list[str], satir: int = 200) -> pd.DataFrame:
    tarihler = pd.date_range("2025-01-01", periods=satir, freq="B")
    sutunlar = pd.MultiIndex.from_product(
        [semboller, ["Open", "High", "Low", "Close", "Volume"]]
    )
    veri = pd.DataFrame(index=tarihler, columns=sutunlar, dtype=float)

    for sira, sembol in enumerate(semboller, start=1):
        taban = np.linspace(100 + sira, 150 + sira, satir)
        veri[(sembol, "Open")] = taban
        veri[(sembol, "High")] = taban + 2
        veri[(sembol, "Low")] = taban - 2
        veri[(sembol, "Close")] = taban + 0.5
        veri[(sembol, "Volume")] = 1_000_000 + sira

    return veri
