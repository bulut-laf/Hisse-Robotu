   1: """Sprint-1: independent common final candidate list (V1).
   2: 
   3: This module deliberately does not import or modify the existing AL/SAT engine,
   4: high-conviction gate, jump-quality gate, or entry-readiness evaluator.
   5: 
   6: Inputs are normalized candidate-like objects or dictionaries. Source adapters
   7: can be wired in a later sprint after this module's contract is validated.
   8: """
   9: from __future__ import annotations
  10: 
  11: from dataclasses import dataclass
  12: from typing import Any, Iterable, Mapping
  13: 
  14: 
  15: SOURCE_PRIORITY = {
  16:     "TEKNIK_TARAMA": 10,
  17:     "FIRSAT_RADARI_HAFTALIK": 20,
  18:     "FIRSAT_RADARI_AYLIK": 30,
  19:     "SICRAMA": 40,
  20: }
  21: 
  22: 
  23: def _value(item: object, name: str, default: Any = None) -> Any:
  24:     if isinstance(item, Mapping):
  25:         return item.get(name, default)
  26:     return getattr(item, name, default)
  27: 
  28: 
  29: def _clean(value: Any) -> str:
  30:     return str(value or "").strip()
  31: 
  32: 
  33: @dataclass(frozen=True)
  34: class CandidateEvidence:
  35:     source: str
  36:     horizon: str = ""
  37:     status: str = ""
  38:     reason: str = ""
  39: 
  40: 
  41: @dataclass(frozen=True)
  42: class CommonFinalCandidate:
  43:     symbol: str
  44:     sources: tuple[str, ...]
  45:     horizons: tuple[str, ...]
  46:     evidence: tuple[CandidateEvidence, ...]
  47:     priority: int
  48:     technical_score: float | None
  49:     volume_ratio: float | None
  50:     risk_reward: float | None
  51:     entry_readiness: str
  52:     data_status: str
  53:     reasons: tuple[str, ...]
  54: 
  55: 
  56: def _number(item: object, name: str) -> float | None:
  57:     raw = _value(item, name)
  58:     if raw is None or raw == "":
  59:         return None
  60:     try:
  61:         return float(raw)
  62:     except (TypeError, ValueError):
  63:         return None
  64: 
  65: 
  66: def build_common_final_candidate_list(
  67:     candidates: Iterable[object],
  68:     *,
  69:     limit: int | None = None,
  70: ) -> tuple[CommonFinalCandidate, ...]:
  71:     """Merge normalized candidates by symbol without generating a new score.
  72: 
  73:     Expected fields:
  74:       symbol (required), source, horizon, status, reason, priority,
  75:       technical_score, volume_ratio, risk_reward, entry_readiness, data_status.
  76: 
  77:     Duplicate symbols are consolidated. Existing numeric values are retained
  78:     as the maximum observed value for technical_score, volume_ratio and
  79:     risk_reward; these are display/context fields, not a combined score.
  80:     Lower numeric priority means earlier in the existing priority convention.
  81:     Missing priority falls back to source order; source precedence only breaks
  82:     ties and never upgrades an underlying AL/SAT decision.
  83:     """
  84:     grouped: dict[str, list[object]] = {}
  85:     for item in candidates:
  86:         symbol = _clean(_value(item, "symbol")).upper()
  87:         if not symbol:
  88:             continue
  89:         grouped.setdefault(symbol, []).append(item)
  90: 
  91:     output: list[CommonFinalCandidate] = []
  92:     for symbol, items in grouped.items():
  93:         sources_seen: list[str] = []
  94:         horizons_seen: list[str] = []
  95:         evidence: list[CandidateEvidence] = []
  96:         reasons: list[str] = []
  97:         priorities: list[int] = []
  98:         scores: list[float] = []
  99:         volumes: list[float] = []
 100:         risk_rewards: list[float] = []
 101:         readiness_values: list[str] = []
 102:         data_values: list[str] = []
 103: 
 104:         for item in items:
 105:             source = _clean(_value(item, "source", "BILINMIYOR")).upper() or "BILINMIYOR"
 106:             horizon = _clean(_value(item, "horizon")).upper()
 107:             status = _clean(_value(item, "status")).upper()
 108:             reason = _clean(_value(item, "reason"))
 109:             if source not in sources_seen:
 110:                 sources_seen.append(source)
 111:             if horizon and horizon not in horizons_seen:
 112:                 horizons_seen.append(horizon)
 113:             evidence.append(CandidateEvidence(source, horizon, status, reason))
 114:             if reason and reason not in reasons:
 115:                 reasons.append(reason)
 116: 
 117:             raw_priority = _value(item, "priority")
 118:             try:
 119:                 if raw_priority is not None and raw_priority != "":
 120:                     priorities.append(int(raw_priority))
 121:             except (TypeError, ValueError):
 122:                 pass
 123: 
 124:             for field, bucket in (
 125:                 ("technical_score", scores),
 126:                 ("volume_ratio", volumes),
 127:                 ("risk_reward", risk_rewards),
 128:             ):
 129:                 value = _number(item, field)
 130:                 if value is not None:
 131:                     bucket.append(value)
 132: 
 133:             readiness = _clean(_value(item, "entry_readiness")).upper()
 134:             if readiness and readiness not in readiness_values:
 135:                 readiness_values.append(readiness)
 136:             data_status = _clean(_value(item, "data_status")).upper()
 137:             if data_status and data_status not in data_values:
 138:                 data_values.append(data_status)
 139: 
 140:         # Use source priority only as a deterministic fallback/tie-breaker.
 141:         fallback_priority = min(
 142:             (SOURCE_PRIORITY.get(s, 999) for s in sources_seen),
 143:             default=999,
 144:         )
 145:         priority = min(priorities) if priorities else fallback_priority
 146:         output.append(
 147:             CommonFinalCandidate(
 148:                 symbol=symbol,
 149:                 sources=tuple(sources_seen),
 150:                 horizons=tuple(horizons_seen),
 151:                 evidence=tuple(evidence),
 152:                 priority=priority,
 153:                 technical_score=max(scores) if scores else None,
 154:                 volume_ratio=max(volumes) if volumes else None,
 155:                 risk_reward=max(risk_rewards) if risk_rewards else None,
 156:                 entry_readiness=" | ".join(readiness_values),
 157:                 data_status=" | ".join(data_values),
 158:                 reasons=tuple(reasons),
 159:             )
 160:         )
 161: 
 162:     output.sort(
 163:         key=lambda c: (
 164:             c.priority,
 165:             -SOURCE_PRIORITY.get(c.sources[0], 0) if c.sources else 0,
 166:             c.symbol,
 167:         )
 168:     )
 169:     if limit is not None:
 170:         if limit < 0:
 171:             raise ValueError("limit must be >= 0 or None")
 172:         output = output[:limit]
 173:     return tuple(output)

  35: def _normalize_ohlcv(df: pd.DataFrame) -> pd.DataFrame:
  36:     x = df.copy()
  37:     if isinstance(x.columns, pd.MultiIndex):
  38:         x.columns = [str(c[0]) for c in x.columns]
  39:     cmap = {str(c).strip().lower(): c for c in x.columns}
  40:     needed = {"open": "Open", "high": "High", "low": "Low", "close": "Close"}
  41:     missing = [src for src in needed if src not in cmap]
  42:     if missing:
  43:         raise ValueError("Eksik OHLC kolonlari: " + ", ".join(missing))
  44:     out = pd.DataFrame(index=pd.to_datetime(x.index))
  45:     for src, dest in needed.items():
  46:         out[dest] = pd.to_numeric(x[cmap[src]], errors="coerce")
  47:     vol = cmap.get("volume")
  48:     out["Volume"] = pd.to_numeric(x[vol], errors="coerce") if vol else np.nan
  49:     out = out.dropna(subset=["Open", "High", "Low", "Close"]).sort_index()
  50:     return out[(out["High"] >= out["Low"]) & (out["Close"] >= out["Low"]) & (out["Close"] <= out["High"])]

  53: def _rsi(s: pd.Series, n: int = 14) -> pd.Series:
  54:     delta = s.diff()
  55:     up = delta.clip(lower=0)
  56:     down = -delta.clip(upper=0)
  57:     au = up.ewm(alpha=1 / n, adjust=False, min_periods=n).mean()
  58:     ad = down.ewm(alpha=1 / n, adjust=False, min_periods=n).mean()
  59:     rs = au / ad.replace(0, np.nan)
  60:     return (100 - 100 / (1 + rs)).fillna(50.0)
