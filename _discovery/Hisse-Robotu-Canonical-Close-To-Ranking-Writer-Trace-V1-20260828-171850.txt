            candidate_count=existing.candidate_count,

        scan_result = engine.scan()

        )
        finished = datetime.now(ISTANBUL).isoformat(timespec="seconds")
        record = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=started_at,
            finished_at=finished,
            status="SUCCESS",
            candidate_count=len(cards),

            run_kind=run_kind,
        )
    state_store.save(record)
    history_store.append(record)
    return record

def append_snapshot(self, observations):
        incoming = tuple(observations)
        if not incoming:
            return 0

        existing_keys = {self._key(item) for item in self.list()}
        new_items = [item for item in incoming if self._key(item) not in existing_keys]
        if not new_items:
            return 0

        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self.path.open("a", encoding="utf-8") as handle:
            for item in new_items:
                handle.write(
                    json.dumps(asdict(item), ensure_ascii=False, sort_keys=True) + "\n"
                )
        return len(new_items)

def observations_from_research_rows(*, rows, as_of, universe_source, profile_name):
    materialized = tuple(rows)
    ranked_universe_size = len(materialized)

    if ranked_universe_size == 0:
        return ()

    observations = []

    for row in materialized:
        symbol = _clean_symbol(
            row.get("Hisse")
            or row.get("hisse")
            or row.get("symbol")
            or row.get("Symbol")
        )

        rank = _coerce_rank(
            row.get("Sıra")
            or row.get("Sira")
            or row.get("rank")
            or row.get("Rank")
        )

        if not symbol or rank is None:
            continue

        observations.append(
            RankObservation(
                as_of=str(as_of),
                symbol=symbol,
                rank=rank,
                ranked_universe_size=ranked_universe_size,
                universe_source=str(universe_source or ""),
                profile_name=str(profile_name or ""),
            )

def _decision_for(card: object) -> tuple[str, tuple[str, ...], tuple[str, ...]]:
    score = int(getattr(card, "technical_score"))
    view = str(getattr(card, "technical_view", ""))
    priority = str(getattr(card, "research_priority", "NORMAL")).upper()
    visibility = str(getattr(card, "visibility_state", "") or "").upper()
    rank_change = getattr(card, "rank_change", None)
    volume = float(getattr(card, "volume_ratio", 0.0))
    rr = float(getattr(card, "risk_reward", 0.0))
    blocked = bool(getattr(card, "data_blocked", False))
    selection_source = str(getattr(card, "selection_source", "TARAMA") or "TARAMA").upper()
    jump_quality = str(getattr(card, "jump_quality", "-") or "-").upper()
    entry_readiness = str(getattr(card, "entry_readiness", "-") or "-").upper()

    reasons: list[str] = []
    cautions: list[str] = list(getattr(card, "risks", ()) or ())

    if _positive_technical(view):
        reasons.append("Teknik görünüm pozitif")
    if score >= 70:
        reasons.append(f"Teknik skor güçlü ({score})")
    if volume >= 1.0:
        reasons.append(f"Hacim teyidi mevcut ({volume:.2f})")
    if rr >= 1.5:
        reasons.append(f"Risk/getiri yeterli ({rr:.2f})")
    if priority == "HIGH":
        reasons.append("Araştırma önceliği yüksek")
    if rank_change is not None and int(rank_change) > 0:
        reasons.append(f"Sıralama iyileşiyor (+{int(rank_change)})")
    if visibility in {"IMPROVING", "RISING", "ACCELERATING"}:
        reasons.append(f"Erken görünürlük {visibility}")

    if blocked:
        cautions.append("Kısmi/gecikmeli veri — karar mevcut son güvenilir veriyle üretildi")
    if volume < 1.0:
        cautions.append(f"Hacim teyidi zayıf ({volume:.2f})")
    if rr < 1.5:
        cautions.append(f"Risk/getiri hedefin altında ({rr:.2f})")
    if rank_change is not None and int(rank_change) < 0:
        cautions.append(f"Sıralama geriliyor ({int(rank_change)})")

    jump_ready_for_buy = True
    if selection_source == "SICRAMA":
        if jump_quality not in {"UYGUN", "-"}:
            jump_ready_for_buy = False
            cautions.append("Sıçrama kalite kapısı AL için uygun değil")
        if entry_readiness not in {"GİRİŞE HAZIR", "-"}:
            jump_ready_for_buy = False
            cautions.append(f"Sıçrama giriş hazırlığı: {entry_readiness}")

    # Closeout yeni birleşik skor üretmez; mevcut kanıtlar için deterministik kapı uygular.
    if _negative_technical(view) or score < 55 or rr < 1.0:
        decision = "İŞLEM YOK"
    elif (
        _positive_technical(view)
        and score >= 70
        and rr >= 1.5
        and volume >= 1.0
        and jump_ready_for_buy
    ):
        decision = "AL"
    else:
        decision = "BEKLE"

    return decision, tuple(reasons), tuple(dict.fromkeys(cautions))

                    )
                )

                hatalar.append(MomentumFailure(sembol=sembol, hata=str(hata)))

        adaylar.sort(
            key=lambda aday: (

def calistir(
        self,
        veriler: dict[str, pd.DataFrame],
        *,
        kesim_tarihi: str,
        ufuklar: Iterable[int] = (5, 10, 20),
        sinyal_gecmisi_olustur: bool = False,
        sinyal_gecmisi_adim_sayisi: int = 6,
        sinyal_gecmisi_adim_bari: int = 5,
    ) -> tuple[tuple[MomentumWalkForwardOutcome, ...], tuple[MomentumFailure, ...]]:
        kesim = pd.Timestamp(kesim_tarihi)
        ufuk = tuple(sorted(set(int(gun) for gun in ufuklar)))
        gecmisler: dict[str, pd.DataFrame] = {}
        tam_veriler: dict[str, pd.DataFrame] = {}
        hatalar: list[MomentumFailure] = []

        for sembol, veri in veriler.items():
            sirali = veri.sort_index().copy()
            tarih_indexi = pd.to_datetime(sirali.index)
            gecmis = sirali.loc[tarih_indexi <= kesim]
            gelecek = sirali.loc[tarih_indexi > kesim]
            if len(gecmis) < 80:
                hatalar.append(
                    MomentumFailure(
                        sembol=sembol,
                        hata=(
                            f"{kesim_tarihi} kesiminde yalnızca {len(gecmis)} "
                            "işlem günü var; en az 80 gerekli."
                        ),
                    )
                )
                continue
            if gelecek.empty:
                hatalar.append(
                    MomentumFailure(
                        sembol=sembol,
                        hata=f"{kesim_tarihi} sonrasında ileri performans verisi yok.",
                    )
                )
                continue
            gecmisler[sembol] = gecmis
            tam_veriler[sembol] = sirali

        adaylar, motor_hatalari = self._motor.analiz_et(gecmisler)
        hatalar.extend(motor_hatalari)
        gecmis_haritasi: dict[str, tuple[MomentumSignalSnapshot, ...]] = {}
        if sinyal_gecmisi_olustur and gecmisler:
            gecmis_haritasi = self._sinyal_gecmisi_olustur(
                gecmisler,
                adim_sayisi=sinyal_gecmisi_adim_sayisi,
                adim_bari=sinyal_gecmisi_adim_bari,
            )

        sonuclar: list[MomentumWalkForwardOutcome] = []
        azami_ufuk = max(ufuk)
        for aday in adaylar:
            sinyal_gecmisi = gecmis_haritasi.get(aday.sembol, ())
            aday = self._gecmisle_zenginlestir(aday, sinyal_gecmisi)
            tam = tam_veriler[aday.sembol]
            ileri = tam.loc[pd.to_datetime(tam.index) > kesim].head(azami_ufuk)
            kapanislar = pd.to_numeric(ileri["Close"], errors="coerce").dropna()
            baslangic = float(aday.kapanis)
            getiriler: list[tuple[int, float | None]] = []
            for gun in ufuk:
                deger = (
                    float(kapanislar.iloc[gun - 1] / baslangic - 1.0)
                    if len(kapanislar) >= gun and baslangic > 0
                    else None
                )
                getiriler.append((gun, deger))

            yuksekler = pd.to_numeric(
                ileri["High"] if "High" in ileri.columns else ileri["Close"],
                errors="coerce",
            ).dropna()
            dusukler = pd.to_numeric(
                ileri["Low"] if "Low" in ileri.columns else ileri["Close"],
                errors="coerce",
            ).dropna()
            if yuksekler.empty or dusukler.empty or baslangic <= 0:
                mfe = None
                mae = None
                maksimum_fiyat = None
                minimum_fiyat = None
                maksimum_tarih = ""
                minimum_tarih = ""
            else:
                maksimum_fiyat = float(yuksekler.max())
                minimum_fiyat = float(dusukler.min())
                mfe = float(maksimum_fiyat / baslangic - 1.0)
                # Azami ters hareket kullanıcıya pozitif risk büyüklüğü olarak gösterilir.
                mae = max(0.0, float(1.0 - minimum_fiyat / baslangic))
                maksimum_tarih = str(pd.Timestamp(yuksekler.idxmax()).date())
                minimum_tarih = str(pd.Timestamp(dusukler.idxmin()).date())

                    ileri_bar_sayisi=len(ileri),
                    maksimum_fiyat=maksimum_fiyat,
                    maksimum_fiyat_tarihi=maksimum_tarih,
                    minimum_fiyat=minimum_fiyat,
                    minimum_fiyat_tarihi=minimum_tarih,
                    sinyal_gecmisi=sinyal_gecmisi,
                )
            )

        sonuclar.sort(
            key=lambda sonuc: (
                _sinif_onceligi(sonuc.aday.sinif),
                -sonuc.aday.momentum_skoru,
                sonuc.aday.sembol,

def _sinyal_gecmisi_olustur(
        self,
        gecmisler: dict[str, pd.DataFrame],
        *,
        adim_sayisi: int,
        adim_bari: int,
    ) -> dict[str, tuple[MomentumSignalSnapshot, ...]]:
        tum_tarihler = sorted(
            {
                pd.Timestamp(tarih)
                for veri in gecmisler.values()
                for tarih in pd.to_datetime(veri.index)
            }
        )
        if not tum_tarihler:
            return {}
        secili_tarihler: list[pd.Timestamp] = []
        son = len(tum_tarihler) - 1
        for sira in range(adim_sayisi - 1, -1, -1):
            indeks = son - sira * adim_bari
            if indeks >= 0:
                secili_tarihler.append(tum_tarihler[indeks])

        harita: dict[str, list[MomentumSignalSnapshot]] = {
            sembol: [] for sembol in gecmisler
        }
        for tarih in secili_tarihler:
            kesit = {
                sembol: veri.loc[pd.to_datetime(veri.index) <= tarih]
                for sembol, veri in gecmisler.items()
                if len(veri.loc[pd.to_datetime(veri.index) <= tarih]) >= 80
            }
            adaylar, _ = self._motor.analiz_et(kesit)
            for aday in adaylar:
                harita.setdefault(aday.sembol, []).append(
                    MomentumSignalSnapshot(
                        tarih=aday.analiz_tarihi,
                        sinif=aday.sinif,
                        momentum_skoru=aday.momentum_skoru,
                        kapanis=aday.kapanis,
                        yakin_direnc=aday.yakin_direnc,
                        kirilim_tetik_seviyesi=aday.kirilim_tetik_seviyesi,
                        dirence_uzaklik=aday.yirmi_gunluk_zirve_uzakligi,
                        izleme_disi_nedeni=aday.izleme_disi_nedeni,
                    )
                )
        return {sembol: tuple(kayitlar) for sembol, kayitlar in harita.items()}

                    else None
                ),

            )
        )

    return tuple(results)

    phase_value = _normalize_phase(phase)
    timestamp = (now or datetime.now(ISTANBUL)).astimezone(ISTANBUL)
    run_date = timestamp.date().isoformat()
    state_store = store or ScheduledRunJsonStore()
    history_store = audit_store or ScheduledRunAuditJsonStore()
    key = f"{run_date}:{phase_value}"
    existing = state_store.load().get(key)
    if existing is not None and existing.status == "SUCCESS" and not force:
        skipped = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=existing.started_at,
            finished_at=existing.finished_at,
            status="SKIPPED",
            candidate_count=existing.candidate_count,
            buy_symbols=existing.buy_symbols,
            detail="Aynı gün/faz için başarılı otomatik koşu zaten mevcut.",
            run_kind=run_kind,
        )
        history_store.append(skipped)
        return skipped

    started_at = timestamp.isoformat(timespec="seconds")
    engine = runtime or _default_runtime()
    try:
        summary = engine.operation_summary()
        scan_result = engine.scan()
        visibility = engine.visibility_contexts()
        candidate_snapshot = engine.candidate_snapshot(summary, scan_result, visibility)
        cards = tuple(getattr(candidate_snapshot, "candidates", ()) or ())
        decisions = engine.decisions(cards)
        engine.persist_decision_snapshot(cards, decisions, phase_value, timestamp)
        buy_symbols = tuple(
            str(getattr(item, "symbol", "")).strip().upper()
            for item in decisions
            if str(getattr(item, "decision", "")).strip().upper() == "AL"
        )
        finished = datetime.now(ISTANBUL).isoformat(timespec="seconds")
        record = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=started_at,
            finished_at=finished,
            status="SUCCESS",
            candidate_count=len(cards),
            buy_symbols=buy_symbols,
            detail="Seans sonu/sabah günlük karar koşusu tamamlandı.",
            run_kind=run_kind,
        )
    except Exception as error:
        finished = datetime.now(ISTANBUL).isoformat(timespec="seconds")
        record = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=started_at,
            finished_at=finished,
            status="FAILED",
            candidate_count=0,
            buy_symbols=(),
            detail=f"{type(error).__name__}: {error}",
            run_kind=run_kind,
        )
    state_store.save(record)
    history_store.append(record)
    return record
