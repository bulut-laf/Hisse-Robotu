def periods_from_csv_bytes(
    payload: bytes,
) -> list[FinancialPeriod]:

        text = payload.decode("utf-8-sig")
    except UnicodeDecodeError as exc:

            "Bilan├зo CSV ba┼Яl─▒─Я─▒ bulunamad─▒."

def build_fundamental_snapshot(
    periods: list[FinancialPeriod],
    *,
    source_metadata: dict[str, str] | None = None,
) -> FundamentalSnapshot:
    if not periods:
        raise ValueError(
            "Temel analiz i├зin en az bir finansal d├╢nem gereklidir."
        )

    symbols = {period.symbol for period in periods}
    currencies = {period.currency for period in periods}
    if len(symbols) != 1:
        raise ValueError(
            "Tek bir temel analiz ├╢zeti yaln─▒zca bir sembol i├зerebilir."
        )
    if len(currencies) != 1:
        raise ValueError(
            "D├╢nemlerin para birimi ayn─▒ olmal─▒d─▒r."
        )

    ordered = sorted(
        periods,
        key=lambda item: (
            item.period_end,
            item.fiscal_year,
            item.fiscal_quarter,
        ),
        reverse=True,
    )
    latest = ordered[0]

    previous_quarter = next(
        (
            item
            for item in ordered[1:]
            if (
                item.fiscal_year,
                item.fiscal_quarter,
            )
            == _previous_quarter_key(
                latest.fiscal_year,
                latest.fiscal_quarter,
            )
        ),
        None,
    )
    previous_year_same_quarter = next(
        (
            item
            for item in ordered[1:]
            if item.fiscal_year == latest.fiscal_year - 1
            and item.fiscal_quarter == latest.fiscal_quarter
        ),
        None,
    )

    ratios = calculate_ratios(latest)

    return FundamentalSnapshot(
        latest=latest,
        previous_quarter=previous_quarter,
        previous_year_same_quarter=previous_year_same_quarter,
        ratios=ratios,
        quarter_growth=(
            calculate_growth(latest, previous_quarter)
            if previous_quarter is not None
            else None
        ),
        annual_growth=(
            calculate_growth(
                latest,
                previous_year_same_quarter,
            )
            if previous_year_same_quarter is not None
            else None
        ),
        quality_flags=_quality_flags(latest, ratios),
        source_metadata=source_metadata or {},
    )
