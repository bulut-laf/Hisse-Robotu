def sembol_eslesmeleri(
    token: str,
    universe_symbols: list[str],
    *,
    company_names: dict[str, str] | None = None,
) -> list[str]:
    q = token.strip().upper().removesuffix(".IS")
    if not q:
        return []

    canonical = [s.removesuffix(".IS") for s in universe_symbols]

    exact = sorted({s for s in canonical if s == q})
    if exact:
        return exact

    starts = sorted({s for s in canonical if s.startswith(q)})
    if starts:
        return starts

    contains = sorted({s for s in canonical if q in s})
    if contains:
        return contains

    if not company_names:
        return []

    normalized_names = {
        symbol.removesuffix(".IS").upper(): str(name).strip().upper()
        for symbol, name in company_names.items()
        if str(symbol).strip() and str(name).strip()
    }

    name_exact = sorted({s for s in canonical if normalized_names.get(s) == q})
    if name_exact:
        return name_exact

    name_starts = sorted({
        s for s in canonical
        if normalized_names.get(s, "").startswith(q)
    })
    if name_starts:
        return name_starts

    return sorted({
        s for s in canonical
        if q in normalized_names.get(s, "")
    })

def bist_bilesen_csvsini_ayristir(
    veri: bytes | str,
    evren_adi: str,
    endeks_kodu: str,
    kaynak: str,
    resmi_kaynaktan_guncel: bool,
) -> BistUniverse:
    """Borsa İstanbul'un noktalı virgüllü bileşen CSV'sini ayrıştırır."""
    metin = veri.decode("utf-8-sig") if isinstance(veri, bytes) else veri
    satirlar = csv.reader(StringIO(metin), delimiter=";")

    semboller: list[str] = []
    gorulenler: set[str] = set()
    tarihler: set[str] = set()

            f"{endeks_kodu} için geçerli hisse bileşeni bulunamadı."

    return BistUniverse(
