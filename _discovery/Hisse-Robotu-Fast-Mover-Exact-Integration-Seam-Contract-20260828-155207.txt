def _preclose_guard(observed_at: datetime) -> None:
    clock = observed_at.timetz().replace(tzinfo=None)
    if clock < PRECLOSE_START:
        raise PreclosePartialDailyPreviewError(
            "17:20 öncesinde pre-close preview çalıştırılamaz."
        )

        raise PreclosePartialDailyPreviewError(
            "18:30 ve sonrasında partial preview yerine canonical close kullanılmalıdır."

def overlay_partial_daily_bar(
    completed_daily: pd.DataFrame,
    snapshot: DelayedDataSnapshot,
    *,
    observed_at: datetime,
) -> pd.DataFrame:
    """Append/replace today's synthetic daily row using current intraday evidence."""

    if completed_daily is None or completed_daily.empty:
        raise PreclosePartialDailyPreviewError("Tamamlanmış günlük seri boş.")

    local_observed = _local_now(observed_at)
    _preclose_guard(local_observed)

    exchange_time = snapshot.exchange_time
    if exchange_time.tzinfo is None:
        exchange_time = exchange_time.replace(tzinfo=ISTANBUL)
    else:
        exchange_time = exchange_time.astimezone(ISTANBUL)

    if exchange_time.date() != local_observed.date():
        raise PreclosePartialDailyPreviewError(
            "Gün içi snapshot bugünkü seansa ait değil."
        )
    if snapshot.locked:
        raise PreclosePartialDailyPreviewError(
            "Gün içi snapshot veri kilidi altında."
        )
    if not snapshot.preview_only:
        # This capability is explicitly for a partial-session preview.
        raise PreclosePartialDailyPreviewError(
            "Pre-close overlay yalnız preview_only intraday snapshot kabul eder."
        )

    required = ("Open", "High", "Low", "Close", "Volume")
    missing = [column for column in required if column not in completed_daily.columns]
    if missing:
        raise PreclosePartialDailyPreviewError(
            "Günlük seride eksik OHLCV sütunları: " + ", ".join(missing)
        )

    values = (
        snapshot.day_open,
        snapshot.day_high,
        snapshot.day_low,
        snapshot.last_price,
        snapshot.cumulative_volume,
    )
    if any(float(value) <= 0 for value in values):
        raise PreclosePartialDailyPreviewError(
            "Partial OHLCV pozitif sayısal değerler içermelidir."
        )

    out = completed_daily.copy()
    today = local_observed.date()
    out = out.loc[_date_index(out) != today].copy()

    synthetic_index = pd.Timestamp(today)
    synthetic = pd.DataFrame(
        {
            "Open": [float(snapshot.day_open)],
            "High": [float(snapshot.day_high)],
            "Low": [float(snapshot.day_low)],
            "Close": [float(snapshot.last_price)],
            "Volume": [float(snapshot.cumulative_volume)],
        },
        index=[synthetic_index],
    )

    # Preserve any extra columns used by an upstream frame without inventing values.
    for column in out.columns:
        if column not in synthetic.columns:
            synthetic[column] = pd.NA
    synthetic = synthetic[out.columns]

    out = pd.concat([out, synthetic], axis=0).sort_index()
    return out

def build_preclose_overlay_fetcher(
    *,
    now: datetime,
    daily_fetcher: Callable[..., BatchMarketDataResult] = toplu_fiyat_verisi_getir,
    intraday_fetcher: Callable[..., ScheduledIntradayBatchResult] = (
        fetch_scheduled_intraday_batch
    ),
) -> Callable[..., BatchMarketDataResult]:
    """Bind a BatchStockScanner-compatible read-only partial-day data fetcher."""

    observed_at = _local_now(now)
    _preclose_guard(observed_at)

    def fetch(
        symbols: Iterable[str],
        *,
        islem_gunu: int = 160,
        indirme_donemi: str = "1y",
        **kwargs: object,
    ) -> BatchMarketDataResult:
        normalized = tuple(str(symbol) for symbol in symbols)

        daily = daily_fetcher(
            normalized,
            islem_gunu=islem_gunu,
            indirme_donemi=indirme_donemi,
            **kwargs,
        )
        intraday = intraday_fetcher(
            normalized,
            now=observed_at,
        )

        overlaid: dict[str, pd.DataFrame] = {}
        failures: list[BatchMarketDataFailure] = list(daily.hatalar)

        intraday_failure_by_symbol = {
            str(item.symbol): str(item.error)
            for item in intraday.failures
        }

        for symbol in normalized:
            daily_frame = daily.veriler.get(symbol)
            if daily_frame is None:
                continue

            snapshot = intraday.snapshots.get(symbol)
            if snapshot is None:
                failures.append(
                    BatchMarketDataFailure(
                        sembol=symbol,
                        hata=(
                            "PRECLOSE_INTRADAY_MISSING: "
                            + intraday_failure_by_symbol.get(
                                symbol,
                                "bugünkü partial snapshot yok",
                            )
                        ),
                    )
                )
                continue

            try:
                overlaid[symbol] = overlay_partial_daily_bar(
                    daily_frame,
                    snapshot,
                    observed_at=observed_at,
                )
            except PreclosePartialDailyPreviewError as error:
                failures.append(
                    BatchMarketDataFailure(
                        sembol=symbol,
                        hata=f"PRECLOSE_OVERLAY_BLOCKED: {error}",
                    )
                )

        # Preserve any additional fields on the canonical BatchMarketDataResult.
        return replace(
            daily,
            veriler=overlaid,
            hatalar=tuple(failures),
        )

    return fetch

def run_preclose_partial_preview(
    *,
    now: datetime | None = None,
    profile_name: str = "dengeli",
    result_limit: int = 50,
    daily_fetcher: Callable[..., BatchMarketDataResult] = toplu_fiyat_verisi_getir,
    intraday_fetcher: Callable[..., ScheduledIntradayBatchResult] = (
        fetch_scheduled_intraday_batch
    ),
) -> PreclosePreviewResult:
    """Run existing BIST scanner with a temporary partial-day overlay.

    The existing scanner, filter profile, BIST universe and StockAnalyzer are
    reused. The result is PREVIEW / NOT CANONICAL and is not persisted by this
    module.
    """

    observed_at = _local_now(now)
    _preclose_guard(observed_at)

    if not 1 <= int(result_limit) <= 100:
        raise PreclosePartialDailyPreviewError(
            "result_limit 1 ile 100 arasında olmalıdır."
        )

    overlay_fetcher = build_preclose_overlay_fetcher(
        now=observed_at,
        daily_fetcher=daily_fetcher,
        intraday_fetcher=intraday_fetcher,
    )
    scanner = BatchStockScanner(veri_getirici=overlay_fetcher)

    service = build_dashboard_market_scanner(
        now_provider=lambda: observed_at,
        tarayici=scanner,
    )

    request_kwargs: dict[str, object] = {
        "evren": "BIST Tüm",
        "profil_adi": profile_name,
        "maksimum_aday": int(result_limit),
        "internetten_evren_guncelle": False,
    }

    # Current production may expose optional sector context. Keep preview lean.
    fields = getattr(MarketScanRequest, "__dataclass_fields__", {})
    if "sektor_baglami_aktif" in fields:
        request_kwargs["sektor_baglami_aktif"] = False

    scan_result = service.calistir(MarketScanRequest(**request_kwargs))

    partial_count = sum(
        1
        for item in tuple(getattr(scan_result, "tum_sonuclar", ()) or ())
        if str(getattr(item, "analiz_tarihi", "")).startswith(
            observed_at.date().isoformat()
        )
    )
    if not partial_count:
        partial_count = sum(
            1
            for item in tuple(
                getattr(getattr(scan_result, "filtrelenmis", None), "adaylar", ())
                or ()
            )
            if str(getattr(item, "analiz_tarihi", "")).startswith(
                observed_at.date().isoformat()
            )
        )

    failure_count = len(
        tuple(getattr(getattr(scan_result, "filtrelenmis", None), "hatalar", ()) or ())
    )

    return PreclosePreviewResult(
        observed_at=observed_at,
        scan_result=scan_result,
        partial_symbol_count=partial_count,
        intraday_failure_count=failure_count,
    )

def fetch_scheduled_intraday_batch(
    symbols: Iterable[str],
    *,
    now: datetime | None = None,
    config: DelayedDataPilotConfig = DelayedDataPilotConfig(),
    chunk_size: int = 40,
    downloader: Callable[..., pd.DataFrame] = yf.download,
) -> ScheduledIntradayBatchResult:
    """Fetch delayed intraday snapshots in chunks using the existing pilot semantics.

    This is a data adapter only. It has no ranking, profile, scanner, signal,
    AL/SAT, or persistence authority.
    """

    normalized = _symbols(symbols)
    if chunk_size < 1:
        raise ValueError("chunk_size en az 1 olmalıdır.")

    received_at = now or datetime.now(ISTANBUL)
    if received_at.tzinfo is None:
        received_at = received_at.replace(tzinfo=ISTANBUL)
    else:
        received_at = received_at.astimezone(ISTANBUL)

    snapshots: dict[str, DelayedDataSnapshot] = {}
    failures: list[ScheduledIntradayBatchFailure] = []

    for start in range(0, len(normalized), chunk_size):
        chunk = normalized[start : start + chunk_size]

        try:
            raw = downloader(
                tickers=list(chunk),
                period=config.period,
                interval=config.interval,
                auto_adjust=True,
                progress=False,
                threads=True,
                prepost=False,
                timeout=45,
                group_by="ticker",
            )
        except Exception as error:
            failures.extend(
                ScheduledIntradayBatchFailure(
                    symbol=symbol,
                    error=f"Toplu gün içi indirme hatası: {error}",
                )
                for symbol in chunk
            )
            continue

        for symbol in chunk:
            try:
                frame = _extract_symbol_frame(
                    raw,
                    symbol,
                    chunk_size=len(chunk),
                )

                def cached_downloader(**_: object) -> pd.DataFrame:
                    return frame

                snapshot = fetch_delayed_snapshot(
                    symbol,
                    config=config,
                    now=received_at,
                    downloader=cached_downloader,
                )
                snapshots[symbol] = snapshot
            except (DelayedDataPilotError, KeyError, TypeError, ValueError) as error:
                failures.append(
                    ScheduledIntradayBatchFailure(
                        symbol=symbol,
                        error=str(error),
                    )
                )

    return ScheduledIntradayBatchResult(
        snapshots=snapshots,
        failures=tuple(failures),
        requested_count=len(normalized),
    )

def _healthy_rank_path(item: RankTrajectory) -> bool:
    values = (
        item.rank_5s,
        item.rank_3s,
        item.rank_1s,
        item.current_rank,
        item.gain_1s,
        item.gain_3s,
        item.gain_5s,

        return False

    return bool(
        item.rank_5s > item.rank_3s > item.rank_1s > item.current_rank
        and item.gain_1s > 0
        and item.gain_3s > 0
        and item.gain_5s > 0
    )

def build_daily_early_trajectory_candidates(
    trajectories: Iterable[RankTrajectory],
    *,
    primary_symbols: Iterable[str] = (),
    limit: int = DEFAULT_EARLY_LIMIT,
    rank_min: int = DEFAULT_EARLY_RANK_MIN,
    rank_max: int = DEFAULT_EARLY_RANK_MAX,
) -> tuple[DailyEarlyTrajectoryCandidate, ...]:
    """
    Read-only early-research selection.

    This is not a score and has no AL/SAT authority. It selects only
    comparable Research trajectories whose 5S -> 3S -> 1S -> today path
    improves monotonically inside the configured research rank band.

    Ordering is deterministic:
    1) 3S rank gain descending (primary early-visibility horizon)
    2) 5S rank gain descending (persistence context)
    3) 1S rank gain descending (latest continuation)
    4) current Research rank ascending
    5) symbol ascending
    """
    if limit < 1:
        raise ValueError("limit 1 veya daha büyük olmalıdır")
    if rank_min < 1 or rank_max < rank_min:
        raise ValueError("rank bandı geçersiz")

    excluded = {
        str(symbol).removesuffix(".IS").upper().strip()
        for symbol in primary_symbols
    }

    candidates: list[DailyEarlyTrajectoryCandidate] = []
    for item in trajectories:
        symbol = str(item.symbol).removesuffix(".IS").upper().strip()
        if symbol in excluded:
            continue
        if not item.comparable:
            continue
        if item.current_rank < rank_min or item.current_rank > rank_max:
            continue
        if not _healthy_rank_path(item):
            continue

        candidates.append(
            DailyEarlyTrajectoryCandidate(
                symbol=symbol,
                current_rank=int(item.current_rank),
                rank_1s=int(item.rank_1s),
                rank_3s=int(item.rank_3s),
                rank_5s=int(item.rank_5s),
                gain_1s=int(item.gain_1s),
                gain_3s=int(item.gain_3s),
                gain_5s=int(item.gain_5s),
            )
        )

    candidates.sort(
        key=lambda item: (
            -item.gain_3s,
            -item.gain_5s,
            -item.gain_1s,
            item.current_rank,
            item.symbol,
        )
    )
    return tuple(candidates[:limit])

def _valid_history(observations: Iterable[RankObservation], symbol: str) -> list[RankObservation]:
    from backend.app.analysis.ranking_trajectory import _parse_observation_date

    rows: list[tuple[object, RankObservation]] = []
    for item in observations:
        if item.symbol != symbol:
            continue
        try:
            parsed = _parse_observation_date(item.as_of)
        except ValueError:
            continue
        rows.append((parsed, item))
    rows.sort(key=lambda pair: pair[0])
    return [item for _, item in rows]

def build_ranking_movement_rows(
    observations: Iterable[RankObservation],
) -> tuple[RankingMovementRow, ...]:
    items = tuple(observations)
    trajectories = build_rank_trajectories(items)
    result: list[RankingMovementRow] = []

    for trajectory in trajectories:
        history = _valid_history(items, trajectory.symbol)
        transitions = max(0, min(5, len(history) - 1))
        positive_steps = _positive_steps_5s(history)
        result.append(
            RankingMovementRow(
                symbol=trajectory.symbol,
                current_rank=trajectory.current_rank,
                gain_1s=trajectory.gain_1s,
                gain_3s=trajectory.gain_3s,
                gain_5s=trajectory.gain_5s,
                acceleration=_acceleration(history),
                positive_steps_5s=positive_steps,
                persistence=_persistence(positive_steps, transitions),
                lifecycle=trajectory.lifecycle,
                top5_distance=max(0, trajectory.current_rank - 5),
                comparable=trajectory.comparable,
            )
        )

    return tuple(
        sorted(
            result,
            key=lambda row: (
                row.gain_3s is None,
                -(row.gain_3s or -10**9),
                -(row.acceleration or -10**9),
                row.current_rank,
                row.symbol,
            ),
        )
    )

def ranking_movement_reason(row: RankingMovementRow) -> str:
    """Ham hareket metriklerinden açıklanabilir görünürlük nedeni üretir."""
    if not row.comparable:
        return "Research source/profile değişti; hareket karşılaştırması bloklu."
    if row.gain_3s is None:
        return "3 seanslık karşılaştırma için yeterli tarihli snapshot yok."
    if row.gain_3s <= 0:
        return "Son 3 seansta pozitif sıra kazanımı yok."

    parts = [f"3S +{row.gain_3s} sıra"]
    if row.gain_1s is not None and row.gain_1s > 0:
        parts.append(f"1S +{row.gain_1s}")
    if row.acceleration is not None and row.acceleration > 0:
        parts.append("hareket hızlanıyor")
    if row.persistence == "KALICI":
        parts.append("5S hareket kalıcı")
    elif row.persistence == "DESTEKLI":
        parts.append("5S hareket destekli")
    elif row.persistence == "TEKIL_SICRAMA":
        parts.append("tekil sıçrama")
    if row.top5_distance <= 20:
        parts.append(f"Top-5'e {row.top5_distance} sıra")
    return " · ".join(parts)
