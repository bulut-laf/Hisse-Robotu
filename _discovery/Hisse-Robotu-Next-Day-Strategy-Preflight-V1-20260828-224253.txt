                f"Geçersiz çıkış nedeni: {self.cikis_nedeni}"
            )

        object.__setattr__(
            self,
            "sembol",
            temiz_sembol,
        )

    @classmethod
    def olustur(
        cls,
        sembol: str,

        cikis_tarihi: date,

        ) - 1
