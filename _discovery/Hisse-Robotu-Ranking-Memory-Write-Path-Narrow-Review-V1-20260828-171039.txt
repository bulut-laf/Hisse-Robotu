def resolve_daily_buy_decision_phase(now: datetime | None = None) -> str | None:
    local = (now or datetime.now(ISTANBUL)).astimezone(ISTANBUL)
    hour = local.hour
    if hour >= 17:
        return "EVENING"
    if 7 <= hour < 12:
        return "MORNING"
    return None

    *,

    normalized = _normalize_phase(phase)

        or result.al_sat_decision_allowed
