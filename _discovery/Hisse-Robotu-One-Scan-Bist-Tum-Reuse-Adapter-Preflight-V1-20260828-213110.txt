def canonical_close(
        self,

def execute_scheduled_run(
    phase: str,
    *,
    now: datetime | None = None,
    runtime: ScheduledRunRuntime | None = None,
    store: ScheduledRunJsonStore | None = None,
    force: bool = False,
    audit_store: ScheduledRunAuditJsonStore | None = None,
    run_kind: str = "SCHEDULED",
) -> ScheduledRunRecord:
    phase_value = _normalize_phase(phase)
    timestamp = (now or datetime.now(ISTANBUL)).astimezone(ISTANBUL)
    run_date = timestamp.date().isoformat()
    state_store = store or ScheduledRunJsonStore()
    history_store = audit_store or ScheduledRunAuditJsonStore()
    key = f"{run_date}:{phase_value}"
    existing = state_store.load().get(key)
    if existing is not None and existing.status == "SUCCESS" and not force:
        skipped = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=existing.started_at,
            finished_at=existing.finished_at,
            status="SKIPPED",
            candidate_count=existing.candidate_count,
            buy_symbols=existing.buy_symbols,
            detail="Aynı gün/faz için başarılı otomatik koşu zaten mevcut.",
            run_kind=run_kind,
        )
        history_store.append(skipped)
        return skipped

    started_at = timestamp.isoformat(timespec="seconds")
    engine = runtime or _default_runtime()
    try:
        summary = engine.operation_summary()
        scan_result = engine.scan()
        visibility = engine.visibility_contexts()
        candidate_snapshot = engine.candidate_snapshot(summary, scan_result, visibility)
        cards = tuple(getattr(candidate_snapshot, "candidates", ()) or ())
        decisions = engine.decisions(cards)
        engine.persist_decision_snapshot(cards, decisions, phase_value, timestamp)
        buy_symbols = tuple(
            str(getattr(item, "symbol", "")).strip().upper()
            for item in decisions
            if str(getattr(item, "decision", "")).strip().upper() == "AL"
        )
        finished = datetime.now(ISTANBUL).isoformat(timespec="seconds")
        record = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=started_at,
            finished_at=finished,
            status="SUCCESS",
            candidate_count=len(cards),
            buy_symbols=buy_symbols,
            detail="Seans sonu/sabah günlük karar koşusu tamamlandı.",
            run_kind=run_kind,
        )
    except Exception as error:
        finished = datetime.now(ISTANBUL).isoformat(timespec="seconds")
        record = ScheduledRunRecord(
            phase=phase_value,
            run_date=run_date,
            started_at=started_at,
            finished_at=finished,
            status="FAILED",
            candidate_count=0,
            buy_symbols=(),
            detail=f"{type(error).__name__}: {error}",
            run_kind=run_kind,
        )
    state_store.save(record)
    history_store.append(record)
    return record

def _default_runtime() -> ScheduledRunRuntime:
    from backend.app.analysis.daily_buy_decision_closeout import build_daily_buy_decisions
    from backend.app.analysis.daily_buy_decision_cycle import (
        DailyBuyDecisionJsonStore,
        build_daily_buy_decision_snapshot,
    )
    from backend.app.analysis.daily_operation_panel import build_daily_operation_summary
    from backend.app.analysis.morning_operation_snapshot import build_morning_operation_snapshot
    from backend.app.analysis.ranking_visibility_service import build_current_visibility_context
    from backend.app.analysis.market_scanner import MarketScanRequest
    from backend.app.compliance.dashboard_market_scanner_source_factory import build_dashboard_market_scanner
    from backend.app.data.delayed_data_pilot import DelayedDataPilotStore
    from backend.app.analysis.signal_outcome_tracking import SignalOutcomeJsonStore
    from backend.app.history.trade_journal import TradeJournalStore, TradeJournalError
    from backend.app.analysis.shadow_portfolio import ShadowPortfolioStore, ShadowPortfolioError

    class Runtime:
        def operation_summary(self) -> object:
            pilot_store = DelayedDataPilotStore()
            signal_store = SignalOutcomeJsonStore()
            trade_store = TradeJournalStore()
            shadow_store = ShadowPortfolioStore()
            try:
                trades = trade_store.list_records()
            except TradeJournalError:
                trades = ()
            try:
                shadow = shadow_store.list()
            except ShadowPortfolioError:
                shadow = ()
            return build_daily_operation_summary(
                pilot_records=pilot_store.list(),
                signal_outcomes=signal_store.list(),
                trade_records=trades,
                shadow_positions=shadow,
            )

        def scan(self) -> object:
            service = build_dashboard_market_scanner(vbts_hard_block=False)
            request = MarketScanRequest(
                evren="BIST 100",
                profil_adi="dengeli",
                maksimum_aday=15,
                ozel_semboller=(),
                internetten_evren_guncelle=True,
                sektor_baglami_aktif=True,
            )
            return service.calistir(request)

        def visibility_contexts(self) -> tuple[object, ...]:
            try:
                return tuple(build_current_visibility_context())
            except Exception:
                return ()

        def candidate_snapshot(self, operation_summary: object, scan_result: object, visibility_contexts: tuple[object, ...]) -> object:
            return build_morning_operation_snapshot(
                operation_summary=operation_summary,
                scan_result=scan_result,
                visibility_contexts=visibility_contexts,
                limit=5,
            )

        def decisions(self, cards: tuple[object, ...]) -> tuple[object, ...]:
            return tuple(build_daily_buy_decisions(cards))

        def persist_decision_snapshot(self, cards: tuple[object, ...], decisions: tuple[object, ...], phase: str, now: datetime) -> None:
            snapshot = build_daily_buy_decision_snapshot(
                cards,
                phase=phase,
                now=now,
                decisions=decisions,
            )
            DailyBuyDecisionJsonStore().save(snapshot)
            from backend.app.analysis.daily_buy_decision_output_archive import (
                DecisionOutputArchiveJsonStore,
            )
            DecisionOutputArchiveJsonStore().save_snapshot(snapshot)

    return Runtime()

def analyze_sector_index(

        loader=loader,
        bars=bars,
    )

    sector_last = pd.Timestamp(sector_frame.index[-1]).date()

def _aggregate_sector(
    definition: SectorMemberDefinition,
    *,
    members: tuple[str, ...],
    benchmark_frame: pd.DataFrame,
    loader: MarketDataLoader,
    frame_cache: dict[str, pd.DataFrame | Exception],
    bars: int,
    minimum_members: int,
    minimum_coverage: float,
) -> SectorMemberStrengthResult:
    benchmark_date = pd.Timestamp(benchmark_frame.index[-1]).date()
    metrics: list[_MemberMetrics] = []

    for symbol in members:
        cached = frame_cache.get(symbol)
        if cached is None:
            try:
                cached = _load_frame(
                    symbol,
                    loader=loader,
                    bars=bars,
                )
            except Exception as exc:
                cached = exc
            frame_cache[symbol] = cached

        if isinstance(cached, Exception):
            continue

        if pd.Timestamp(cached.index[-1]).date() != benchmark_date:
            continue

        try:
            metrics.append(_member_metrics(cached))
        except Exception:
            continue

    total = len(members)
    valid = len(metrics)
    coverage = (valid / total) if total else 0.0

    # BIST 100 kesişiminde bazı gerçek sektörlerin doğal olarak yalnızca
    # iki üyesi bulunabilir. Böyle bir sektörü 2/2 güncel olduğunda tamamen
    # kaybetmek yerine "Düşük" güvenle bağlam olarak tutuyoruz.
    # Tek üyeli sektörler sektör temsili sayılmaz.
    if total <= 1:
        raise SectorMemberStrengthError(
            f"{definition.code} için sektör temsili yetersiz: "
            f"{total} referans üye."
        )

    if total == 2:
        required = 2
        confidence = "Düşük"
    else:
        required = max(
            minimum_members,
            ceil(total * minimum_coverage),
        )
        confidence = (
            "Yüksek"
            if coverage >= 0.80 and valid >= 5
            else "Orta"
        )

    if valid < required:
        raise SectorMemberStrengthError(
            f"{definition.code} için temsil gücü yetersiz: "
            f"{valid}/{total} güncel üye; gereken en az {required}."
        )

    return_1d = float(median(m.return_1d for m in metrics))
    return_5d = float(median(m.return_5d for m in metrics))
    return_20d = float(median(m.return_20d for m in metrics))

    benchmark_close = benchmark_frame["Close"]
    benchmark_return_1d = _return_percent(benchmark_close, 1)
    benchmark_return_5d = _return_percent(benchmark_close, 5)
    benchmark_return_20d = _return_percent(benchmark_close, 20)

    relative_1d = return_1d - benchmark_return_1d
    relative_5d = return_5d - benchmark_return_5d
    relative_20d = return_20d - benchmark_return_20d
    relative_composite = (
        relative_1d * 0.20
        + relative_5d * 0.35
        + relative_20d * 0.45
    )

    positive_1d_ratio = sum(m.positive_1d for m in metrics) / valid
    positive_5d_ratio = sum(m.positive_5d for m in metrics) / valid
    above_ema21_ratio = sum(m.above_ema21 for m in metrics) / valid
    short_trend_ratio = (
        sum(m.short_trend_positive for m in metrics) / valid
    )
    momentum_positive_ratio = (
        sum(m.momentum_positive for m in metrics) / valid
    )

    volume_values = [
        m.volume_ratio
        for m in metrics
        if m.volume_ratio is not None
    ]
    median_volume_ratio = (
        float(median(volume_values))
        if volume_values
        else None
    )

    relative_score = _relative_score(
        relative_1d,
        relative_5d,
        relative_20d,
    )
    trend_score = int(
        round(
            15 * above_ema21_ratio
            + 10 * short_trend_ratio
        )
    )
    momentum_score = int(
        round(
            10 * positive_5d_ratio
            + 10 * momentum_positive_ratio
        )
    )

    volume_component = 0.5
    if median_volume_ratio is not None:
        volume_component = min(
            max(median_volume_ratio / 1.20, 0.0),
            1.0,
        )

    participation_score = int(
        round(
            10 * positive_1d_ratio
            + 5 * volume_component
        )
    )

    strength_score = max(
        0,
        min(
            100,
            relative_score
            + trend_score
            + momentum_score
            + participation_score,
        ),
    )

    return SectorMemberStrengthResult(
        rank=0,
        code=definition.code,
        name=definition.name,
        analysis_date=benchmark_date,
        member_count=total,
        valid_member_count=valid,
        coverage_ratio=coverage,
        return_1d=return_1d,
        return_5d=return_5d,
        return_20d=return_20d,
        benchmark_return_1d=benchmark_return_1d,
        benchmark_return_5d=benchmark_return_5d,
        benchmark_return_20d=benchmark_return_20d,
        relative_1d=relative_1d,
        relative_5d=relative_5d,
        relative_20d=relative_20d,
        relative_composite=relative_composite,
        positive_1d_ratio=positive_1d_ratio,
        positive_5d_ratio=positive_5d_ratio,
        above_ema21_ratio=above_ema21_ratio,
        short_trend_ratio=short_trend_ratio,
        momentum_positive_ratio=momentum_positive_ratio,
        median_volume_ratio=median_volume_ratio,
        relative_score=relative_score,
        trend_score=trend_score,
        momentum_score=momentum_score,
        participation_score=participation_score,
        strength_score=strength_score,
        status=_status(strength_score),
        confidence=confidence,
    )
